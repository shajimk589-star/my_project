import torch
import torch.nn as nn
import torchvision.models as models

class DeepfakeDetectorCNN(nn.Module):
    def __init__(self):
        super(DeepfakeDetectorCNN, self).__init__()
        self.conv1 = nn.Conv2d(3, 32, kernel_size=3, padding=1)
        self.bn1 = nn.BatchNorm2d(32)
        self.conv2 = nn.Conv2d(32, 64, kernel_size=3, padding=1)
        self.bn2 = nn.BatchNorm2d(64)
        self.conv3 = nn.Conv2d(64, 128, kernel_size=3, padding=1)
        self.bn3 = nn.BatchNorm2d(128)
        self.conv4 = nn.Conv2d(128, 256, kernel_size=3, padding=1)
        self.bn4 = nn.BatchNorm2d(256)
        self.pool = nn.MaxPool2d(2, 2)
        self.relu = nn.ReLU()
        self.adaptive_pool = nn.AdaptiveAvgPool2d((7, 7))
        
        self.fc1 = nn.Linear(256 * 7 * 7, 512)
        self.dropout1 = nn.Dropout(0.5)
        self.fc2 = nn.Linear(512, 128)
        self.dropout2 = nn.Dropout(0.3)
        self.fc3 = nn.Linear(128, 1)

    def forward(self, x):
        x = self.pool(self.relu(self.bn1(self.conv1(x))))
        x = self.pool(self.relu(self.bn2(self.conv2(x))))
        x = self.pool(self.relu(self.bn3(self.conv3(x))))
        x = self.pool(self.relu(self.bn4(self.conv4(x))))
        x = self.adaptive_pool(x)
        x = x.view(x.size(0), -1)
        x = self.relu(self.fc1(x))
        x = self.dropout1(x)
        x = self.relu(self.fc2(x))
        x = self.dropout2(x)
        x = self.fc3(x)
        return x

class DeepfakeDetectorResNet(nn.Module):
    def __init__(self, pretrained=True):
        super(DeepfakeDetectorResNet, self).__init__()
        try:
            self.resnet = models.resnet18(weights=models.ResNet18_Weights.DEFAULT if pretrained else None)
        except AttributeError:
            self.resnet = models.resnet18(pretrained=pretrained)
        num_ftrs = self.resnet.fc.in_features
        self.resnet.fc = nn.Sequential(
            nn.Linear(num_ftrs, 128),
            nn.ReLU(),
            nn.Dropout(0.4),
            nn.Linear(128, 1)
        )

    def forward(self, x):
        return self.resnet(x)

%%writefile inference.py
import os
import cv2
import numpy as np
from PIL import Image
import torch
import torchvision.transforms as transforms

from model import DeepfakeDetectorCNN, DeepfakeDetectorResNet

USE_MEDIAPIPE = False
face_detector = None
face_cascade = None

try:
    import mediapipe as mp
    if hasattr(mp, 'solutions') and hasattr(mp.solutions, 'face_detection'):
        mp_face_detection = mp.solutions.face_detection
        face_detector = mp_face_detection.FaceDetection(min_detection_confidence=0.5)
        USE_MEDIAPIPE = True
        print("MediaPipe face detection initialized successfully.")
    else:
        raise AttributeError("mediapipe.solutions.face_detection not available")
except Exception as e:
    print(f"MediaPipe load failed ({e}). Falling back to OpenCV Haar Cascades...")
    cascade_path = cv2.data.haarcascades + 'haarcascade_frontalface_default.xml'
    face_cascade = cv2.CascadeClassifier(cascade_path)

img_transform = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.ToTensor(),
    transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])
])

def load_detector_model(model_type='resnet', device='cpu'):
    model_path = f"model_{model_type}.pth"
    if not os.path.exists(model_path): return None
    model = DeepfakeDetectorCNN() if model_type == 'custom' else DeepfakeDetectorResNet(pretrained=False)
    try:
        model.load_state_dict(torch.load(model_path, map_location=device))
        model.to(device).eval()
        return model
    except Exception as e:
        print(f"Error loading {model_path}: {e}")
        return None

def detect_faces_in_frame(frame):
    h, w, _ = frame.shape
    faces = []
    
    if USE_MEDIAPIPE and face_detector is not None:
        try:
            rgb_frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
            results = face_detector.process(rgb_frame)
            if results.detections:
                for detection in results.detections:
                    bbox = detection.location_data.relative_bounding_box
                    xmin = int(bbox.xmin * w)
                    ymin = int(bbox.ymin * h)
                    width = int(bbox.width * w)
                    height = int(bbox.height * h)
                    
                    pad_w, pad_h = int(width * 0.15), int(height * 0.15)
                    xmin = max(0, xmin - pad_w)
                    ymin = max(0, ymin - pad_h)
                    width = min(w - xmin, width + 2 * pad_w)
                    height = min(h - ymin, height + 2 * pad_h)
                    
                    if width > 10 and height > 10:
                        faces.append((xmin, ymin, width, height))
                return faces
        except Exception as e:
            print(f"MediaPipe error: {e}. Switching to OpenCV fallback...")
            
    if face_cascade is not None and not face_cascade.empty():
        try:
            gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
            detected_faces = face_cascade.detectMultiScale(gray, 1.1, 5, minSize=(30,30))
            for (x, y, width, height) in detected_faces:
                pad_w, pad_h = int(width * 0.1), int(height * 0.1)
                xmin = max(0, x - pad_w)
                ymin = max(0, y - pad_h)
                face_w = min(w - xmin, width + 2 * pad_w)
                face_h = min(h - ymin, height + 2 * pad_h)
                if face_w > 10 and face_h > 10:
                    faces.append((xmin, ymin, face_w, face_h))
        except Exception as e:
            print(f"OpenCV detection error: {e}")
    return faces

def analyze_video(video_path, model, model_type='resnet', sample_fps=2, device='cpu'):
    cap = cv2.VideoCapture(video_path)
    if not cap.isOpened(): return {"error": "Could not open video."}
    
    fps = cap.get(cv2.CAP_PROP_FPS)
    total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
    duration = total_frames / fps if fps > 0 else 0
    sample_step = max(1, int(fps / sample_fps)) if fps > 0 else 15
    
    frame_idx, analyzed_frames_count = 0, 0
    timeline_data, detected_faces_gallery, face_probabilities = [], [], []
    is_fallback = (model is None)
    
    while cap.isOpened():
        ret, frame = cap.read()
        if not ret: break
        if frame_idx % sample_step == 0:
            timestamp = frame_idx / fps if fps > 0 else 0.0
            face_boxes = detect_faces_in_frame(frame)
            if len(face_boxes) > 0:
                frame_rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
                for x, y, w, h in face_boxes:
                    face_crop = frame_rgb[y:y+h, x:x+w]
                    face_pil = Image.fromarray(face_crop)
                    if not is_fallback:
                        tensor = img_transform(face_pil).unsqueeze(0).to(device)
                        with torch.no_grad():
                            prob = torch.sigmoid(model(tensor)).item()
                    else:
                        prob = float(np.sin(x + y * 0.1) * 0.45 + 0.5)
                        prob = np.clip(prob + np.random.uniform(-0.1, 0.1), 0.01, 0.99)
                    face_probabilities.append(prob)
                    timeline_data.append((timestamp, prob))
                    detected_faces_gallery.append({
                        'image': face_pil,
                        'score': prob,
                        'label': 'AI/Fake' if prob >= 0.5 else 'Real',
                        'timestamp': round(timestamp, 2)
                    })
            analyzed_frames_count += 1
        frame_idx += 1
    cap.release()
    
    if len(face_probabilities) == 0:
        return {
            "status": "No Faces Detected",
            "message": "We could not find any faces in this video.",
            "duration": round(duration, 2),
            "analyzed_frames": analyzed_frames_count,
            "overall_score": 0.0,
            "prediction": "N/A",
            "timeline": [],
            "gallery": [],
            "is_fallback": is_fallback
        }
        
    avg_score = float(np.mean(face_probabilities))
    timeline_list = [{"time": round(t, 2), "score": float(s)} for t, s in sorted(timeline_data, key=lambda x: x[0])]
    gallery_sorted = sorted(detected_faces_gallery, key=lambda x: x['score'], reverse=True)
    return {
        "status": "Success",
        "duration": round(duration, 2),
        "analyzed_frames": analyzed_frames_count,
        "overall_score": avg_score,
        "prediction": "AI/Fake" if avg_score >= 0.5 else "Real",
        "timeline": timeline_list,
        "gallery": gallery_sorted,
        "is_fallback": is_fallback
    }

custom_css = """
.gradio-container { background-color: #0b0f19 !important; font-family: 'Inter', sans-serif !important; color: #e2e8f0 !important; }
.header-box { text-align: center; padding: 2rem 1rem; background: linear-gradient(135deg, #1e1b4b 0%, #0f172a 100%); border-radius: 12px; border: 1px solid #3b82f6; margin-bottom: 2rem; }
.header-box h1 { font-size: 2.2rem !important; background: linear-gradient(to right, #60a5fa, #c084fc); -webkit-background-clip: text; -webkit-text-fill-color: transparent; }
.status-verdict { padding: 1.5rem; border-radius: 8px; font-weight: 700; text-align: center; font-size: 1.4rem; margin-bottom: 1.5rem; }
.status-fake { background-color: rgba(239, 68, 68, 0.15) !important; border: 2px solid #ef4444 !important; color: #fca5a5 !important; }

.status-none { background-color: rgba(107, 114, 128, 0.15) !important; border: 2px solid #6b7280 !important; color: #d1d5db !important; }
"""

def create_timeline_plot(timeline):
    if not timeline: return None
    times = [item["time"] for item in timeline]
    scores = [item["score"] * 100 for item in timeline]
    
    plt.style.use('dark_background')
    fig, ax = plt.subplots(figsize=(7, 3.5), facecolor='#0b0f19')
    ax.set_facecolor('#0b0f19')
    
    ax.plot(times, scores, color='#60a5fa', linewidth=2.5, marker='o', markersize=4, label='AI/Fake Probability')
    ax.fill_between(times, scores, color='#60a5fa', alpha=0.15)
    ax.axhline(50, color='#ef4444', linestyle='--', alpha=0.7, label='Threat Level (50%)')
    
    ax.set_title("Temporal Threat Analysis", fontsize=11, color='#e2e8f0', pad=12, fontweight='bold')
    ax.set_xlabel("Time (s)", fontsize=9, color='#94a3b8')
    ax.set_ylabel("Fake Probability (%)", fontsize=9, color='#94a3b8')
    ax.set_ylim(-5, 105)
    
    ax.spines['top'].set_visible(False)
    ax.spines['right'].set_visible(False)
    ax.spines['left'].set_color('#374151')
    ax.spines['bottom'].set_color('#374151')
    ax.tick_params(colors='#94a3b8', labelsize=8)
    ax.grid(True, color='#1e293b', linestyle=':', alpha=0.6)
    ax.legend(facecolor='#111827', edgecolor='#374151', labelcolor='#e2e8f0', fontsize=8)
    
    plt.tight_layout()
    return fig

def predict_video(video_path, model_type):
    if not video_path:
        return "<div class='status-verdict status-none'>No video uploaded.</div>", None, gr.update(visible=False), []
    global loaded_models
    model_file = f'model_{model_type}.pth'
    if loaded_models[model_type] is None and os.path.exists(model_file):
        loaded_models[model_type] = load_detector_model(model_type, device)
    model = loaded_models[model_type]
    
    result = analyze_video(video_path, model, model_type, sample_fps=2, device=device)
    if "error" in result:
        return f"<div class='status-verdict status-fake'>Error: {result['error']}</div>", None, gr.update(visible=False), []
    if result["status"] == "No Faces Detected":
        return f"<div class='status-verdict status-none'>🔍 NO FACES DETECTED</div><p style='text-align:center;'>{result['message']}</p>", None, gr.update(visible=False), []
    
    score_pct = result["overall_score"] * 100
    prediction = result["prediction"]
    is_fallback = result.get("is_fallback", False)
    
    mock_tag = "<br><span style='font-size:0.8rem;color:#fbbf24;'>⚠️ MODEL WEIGHTS NOT FOUND! Running on coordinate-based mock math fallback. Train model first.</span>" if is_fallback else ""
    
    if prediction == "AI/Fake":
        verdict_class = "status-fake"
        verdict_text = f"⚠️ DEEPFAKE / AI DETECTED ({score_pct:.1f}% Fake Confidence)"
    else:
        verdict_class = "status-real"
        verdict_text = f"✅ REAL / GENUINE VIDEO ({100 - score_pct:.1f}% Real Confidence)"
        
    html_output = f"<div class='status-verdict {verdict_class}'>{verdict_text}{mock_tag}</div><div style='margin-top:10px;'><b>Duration:</b> {result['duration']}s | <b>Frames Inspected:</b> {result['analyzed_frames']} | <b>Faces Cropped:</b> {len(result['gallery'])}</div>"
    
    fig = create_timeline_plot(result["timeline"])
    chart_visible = gr.update(visible=True) if fig is not None else gr.update(visible=False)
    
    gallery_data = [(face["image"], f"{face['timestamp']}s | {face['score']*100:.1f}% Fake") for face in result["gallery"]]
    return html_output, fig, chart_visible, gallery_data

with gr.Blocks(css=custom_css, title="AegisEye - Deepfake Video Detector") as demo:
    gr.HTML("<div class='header-box'><h1>🛡️ AegisEye: Deepfake & AI Video Detection</h1><p>A PyTorch ResNet-18 pipeline with Face Extraction</p></div>")
    with gr.Row():
        with gr.Column(scale=1):
            gr.Markdown("### 📥 Input Video")
            video_input = gr.Video(label="Upload Video", sources=["upload"])
            model_type = gr.Dropdown(choices=["custom", "resnet"], value="resnet", label="Model Choice")
            analyze_btn = gr.Button("🔍 Scan Video", variant="primary")
        with gr.Column(scale=1):
            gr.Markdown("### 📊 Results")
            verdict_output = gr.HTML("<div class='status-verdict status-none'>Waiting for upload...</div>")
            timeline_chart = gr.Plot(label="Temporal Threat Analysis", visible=False)
    with gr.Row():
        with gr.Column():
            gr.Markdown("### 🔍 Extracted Faces Visual Gallery")
            gallery_output = gr.Gallery(label="Faces Details", columns=6, rows=2)

    analyze_btn.click(
        fn=predict_video,
        inputs=[video_input, model_type],
        outputs=[verdict_output, timeline_chart, timeline_chart, gallery_output]
    )
